\section{Problem Setting}

Following the standard RBF-FD methods, let $u \in \mathcal{F} = W^{s,p}$ be a solution for a PDE in form of:
\begin{equation}
	- Lu = -\diver\!\big(A \grad u\big) = f \quad \text{in } \Omega,
	\label{eq:PDE}
\end{equation}
Throughout this document, we require that $L$ be a self-adjointed, that is, $L=L^*$. Let $1<p<\infty$, $A \in W^{1,\infty} (\bar{\Omega}; \mathbb{R}^{d \times d}):\bar{\Omega} \rightarrow \mathbb{R}^{d \times d}$ is a symmetric positive definite anisotropic tensor, where
\begin{equation}
	\zeta |\xi|^2 \le \bf{A}(\bf{x}) \xi \cdot \xi \le Z |\xi|^2 \text{, } 0 < \zeta \le Z
\end{equation}
Assume $\Omega \subset \mathbb{R}^d$ and $f \in W^{s-k,p}(\Omega)$ is the forcing term acting in the domain $\Omega$. For $2^{nd}$ order elliptic operators, we set $k=2$. The problem is subject to Neumann and Dirichlet Boundary Conditions partitioned within $\partial \Omega$ of the form:
\begin{align}
  u &= g_D \quad \text{on } \Gamma_D \subset \partial\Omega, \\
  n \cdot A \grad u &= g_N \quad
  \text{on } \Gamma_N \subset \partial\Omega,
  \label{eq:neumann-bc}
\end{align}
For simplicity, we assume $\Omega$ be a bounded $C^{1,1}(\partial \Omega)$ domain, with $\partial \Omega = \Gamma_D \cup \Gamma_N$ and $\Gamma_D \cap \Gamma_N = \emptyset$. Furthermore, let $g_D \in W^{s-1/p,p}(\partial \Omega)$, $g_N \in W^{s-1-1/p,p}(\partial \Omega)$. By \textit{Agmon-Douglis-Nirenberg} Theorem \cite{Agmon}, we have the following estimate for $u$:
\begin{equation}
\|u\|_{W^{s,p}(\Omega)} \leq C\left(\|Lu\|_{W^{s-k,p}(\Omega)} + \|g_D\|_{W^{s-1/p,p}(\partial\Omega)} + \|g_N\|_{W^{s-1-1/p,p}(\partial\Omega)} + \|u\|_{L^p(\Omega)} \right).
\end{equation}
In the case that we work on a Hilbert space, we then have for $p=2$ ($W^{s,2} := H^s$) the following estimate for $u$:
\begin{align*}
\|u\|_{H^s(\Omega)} \leq C\left(\|Lu\|_{H^{s-k}(\Omega)} + \|g_D\|_{H^{s-\frac{1}{2}}(\partial\Omega)} + \|g_N\|_{H^{s-\frac{3}{2}}(\partial\Omega)} + \|u\|_{L^2(\Omega)} \right).
\end{align*}
If we want to study cases where $(Lu)(\bm{x})$ is known pointwise, we can formulate the setup by having $f \in W^{s-k,p}(\Omega)$ and, by Morrey Embedding, 
\begin{equation*}
	W^{s-k,p}(\Omega) \hookrightarrow C^{\mu,\gamma}(\Omega) \hookrightarrow C^{0}(\Omega) \text{, with } \mu := \left \lfloor s-k-\frac{d}{p} \right \rfloor \text{, } \gamma := s-k-\mu-\frac{d}{p}
\end{equation*}
In other words, $s > k + \frac{d}{p}$. Suppose $p=2$ and $k=2$, then $s > 2 + \frac{d}{2}$. Given an operator $L: H^{s}(\Omega) \rightarrow H^{s-2}(\Omega)$, we define an error functional of the form:
\begin{equation*}
	\xi_{L,x}: H^{s}(\Omega) \rightarrow \mathbb{R} \text{, } \xi_{L,x} \in \left(H^{s}\right)^*(\Omega)
\end{equation*}
Then by Riesz Representation, there exists a $v_{L,x} \in H^{s}(\Omega)$ such that $\xi_{L,x} (u) = \left\langle u , v_{L,x} \right\rangle_{H^{s}(\Omega)}$, and $\| \xi_{L,x} |\|_{\left(H^{s}(\Omega)\right)^*} = \| v_{L,x} \|_{H^{s}(\Omega)}$

\section{Functional Analysis Aspect}

We examine the structural framework that governs the RBF-FD method. Let $\mathcal{N}(\Omega):=W^{s,p}(\Omega)$ be the native space whose vectors are point-wise bounded. We later assume that $\mathcal{N}(\Omega)$ is a Hilbert space (this holds, for instance, when $\mathcal{N}(\Omega)$ is the native space of a symmetric positive definite kernel). Let $L:\mathcal{N}(\Omega) \rightarrow W^{s-k,p}(\Omega)$ be a bounded linear operator. For $n \in \mathbb{N}$, let $\theta_1, \dots, \theta_n \in \mathcal{L}\bigl(\mathcal{N}(\Omega), W^{s-k,p}(\Omega)\bigr)$ be bounded linear maps. Define $L_w:\mathcal{N}(\Omega) \rightarrow W^{s-k,p}(\Omega)$ by
\begin{equation}
	L_w u := \sum_{i=1}^n \theta_i(u).
\end{equation}
This operator is linear and bounded, since
\begin{equation*}
	\| L_w u \|_{W^{s-k,p}(\Omega)} \le \| \bm{w} \|_{l^2} \, \| \theta \|_{\mathcal{L}\bigl(\mathcal{N}(\Omega), W^{s-k,p}(\Omega)\bigr)} \, \| u \|_{\mathcal{N}(\Omega)}.
\end{equation*}
The goal is for $L_w$ to approximate $L$.

We further assume that there is a linear subspace $\mathcal{V}_h(\Omega) \subseteq \mathcal{N}(\Omega)$ on which $L_w$ reproduces $L$ exactly, i.e., $Lu = L_w u$ for all $u \in \mathcal{V}_w(\Omega)$. We define the error operator $\xi_{w,L} \in \mathcal{L}\bigl(\mathcal{N}(\Omega), W^{s-k,p}(\Omega)\bigr)$ by
\begin{equation*}
	\xi_{w,L}(u) := Lu - L_w u .
\end{equation*}
It is linear and bounded because $L$ and $L_w$ are, and it vanishes on $\mathcal{V}_w(\Omega)$. Let $1 < p < \infty$ and let $q$ satisfy $\frac{1}{p} + \frac{1}{q} = 1$. For $g \in L^{q}(\Omega)$ define the linear functional $l_g$ on $W^{s-k,p}(\Omega)$ by
\begin{equation*}
	l_g(v) := \sum_{|\alpha| \le k} \int_\Omega v \, g \, d\bm{x}.
\end{equation*}
With $\| v \|_{W^{s-k,p}(\Omega)} = \bigl( \sum_{|\alpha| \le k} \| D^\alpha v \|_{L^p(\Omega)}^p \bigr)^{1/p}$, the H\"{o}lder inequality (applied to the integrals and then to the finite sum over $\alpha$) gives
\begin{equation*}
	|l_g(v)| \le \| v \|_{W^{s-k,p}(\Omega)} \| g \|_{L^{q}(\Omega)}.
\end{equation*}
Set $\xi_{w,L,g} := l_g \circ \xi_{w,L}$. It is linear, and for every $u \in \mathcal{N}(\Omega)$
\begin{align*}
	|\xi_{w,L,g}(u)| &\le \| \xi_{w,L}(u) \|_{W^{s-k,p}(\Omega)} \| g \|_{L{^q}(\Omega)}\\
	&\le \| \xi_{w,L} \|_{\mathcal{L}(\mathcal{N}, W^{s-k,p})} \| g \|_{L^{q}(\Omega)} \| u \|_{\mathcal{N}(\Omega)}.
\end{align*}
Assuming $\xi_{w,L}$ is bounded, then $\xi_{w,L,g}$ is a bounded functional, $\xi_{w,L,g} \in \mathcal{N}(\Omega)^*$. Furthermore, if $\mathcal{N}(\Omega)$ is a Hilbert space, the Riesz Representation Theorem gives a unique $v_{w,L,g} \in \mathcal{N}(\Omega)$ such that $\xi_{w,L,g}(u) = \left\langle u, v_{w,L,g} \right\rangle_{\mathcal{N}(\Omega)}$ for all $u \in \mathcal{N}(\Omega)$. Moreover,
\begin{align*}
	\| \xi_{w,L,g} \|_{\mathcal{N}^*(\Omega)} &= \sup_{\substack{u \in \mathcal{N}(\Omega) \\ \| u \|_{\mathcal{N}(\Omega)} \le 1}} |\xi_{w,L,g}(u)| \\
	&= \sup_{\substack{u \in \mathcal{N}(\Omega) \\ u \ne 0}} \frac{|\xi_{w,L,g}(u)|}{\| u \|_{\mathcal{N}(\Omega)}} \\
	&= \sup_{\substack{u \in \mathcal{N}(\Omega) \\ u \ne 0}} \frac{|\left\langle u, v_{w,L,g} \right\rangle_{\mathcal{N}(\Omega)}|}{\| u \|_{\mathcal{N}(\Omega)}} = \| v_{w,L,g} \|_{\mathcal{N}(\Omega)}.
\end{align*}

\subsection{Pointwise Evaluation}

Suppose $s > k + \frac{d}{p}$. By Morrey Embedding, $W^{s-k,p}(\Omega) \hookrightarrow C^{\mu,\gamma}(\Omega) \hookrightarrow C^{0}(\Omega)$  Then we have a bounded functional $\delta_z \in \left(W^{s-k,p}(\Omega) \right)^*$:
\begin{equation}
	\xi_{w,L}(u)(z) = \delta_z \left( \xi_{w,L} (u) \right) := \xi_{w,L,z} (u)
\end{equation}
Then $\xi_{w,L,z} \in \mathcal{N}^*(\Omega)$ given $\xi_{w,L}$ is bounded. Furthermore, Assuming again $\mathcal{N}(\Omega)$ is Hilbert, then there exists a representer $v_{w,L,z}$ such that
\begin{align*}
	\| \xi_{w,L,z} \|_{\mathcal{N}(\Omega)^*} &= \sup_{\substack{u \in \mathcal{N}(\Omega) \\ \| u \|_{\mathcal{N}(\Omega)} \le 1}} |\xi_{w,L,z}(u)| \\
	&= \sup_{\substack{u \in \mathcal{N}(\Omega) \\ u \ne 0}} \frac{|\xi_{w,L,z}(u)|}{\| u \|_{\mathcal{N}(\Omega)}} \\
	&= \sup_{\substack{u \in \mathcal{N}(\Omega) \\ u \ne 0}} \frac{|\left\langle u, v_{w,L,z} \right\rangle_{\mathcal{N}(\Omega)}|}{\| u \|_{\mathcal{N}(\Omega)}} = \| v_{w,L,z} \|_{\mathcal{N}(\Omega)}.
\end{align*}
We can concretely define $\xi_{w,L,z}$ as follows:
\begin{align*}
	\xi_{w,L,z}(u) &= Lu(z) - \sum_{i=1}^n \delta_z \circ\theta_i(u)\\
	\xi_{w,L,z}(u) &= Lu(z) - \sum_{i=1}^n w_{i,z} u(x_i)
\end{align*}
Here we set $\delta_z \circ \theta_{i} = w_{i,z} \delta_{x_i}$, with $w_{1,z}, \hdots, w_{n,z} \in \mathbb{R}$.

\section{RBF-FD Method}

\subsection{Local Approximation setup}
To obtain an approximation of the operator $L$ at a point $\bm{z} \in \Omega$, we approximate $u$ by a function $\hat{u}$ on a neighborhood $\Omega_{\bm{z}} \subseteq \Omega$ of $\bm{z}$, so that $Lu(\bm{z}) \approx L\hat{u}(\bm{z})$. Let $Y = \{ \bm{y}_1, \dots, \bm{y}_m \} \subseteq \Omega_{\bm{z}}$ be a set of centers, let $\phi_i:\Omega_{\bm{z}} \rightarrow \mathbb{R}$, $i = 1, \dots, m$, be basis functions, and define the vector space $V_Y = \operatorname{span}\left\{ \phi_1, \dots, \phi_m \right\}$. Let $X = \left\{ \bm{x}_1, \dots, \bm{x}_n \right\} \subseteq \Omega_{\bm{z}}$ be the set of stencil nodes for $\bm{z}$. For the classical approximation scheme, let $m=n$. We define $\bm{\Phi} \in \mathbb{R}^{n \times m}$ by $\bm{\Phi}_{ij} = \phi_j(\bm{x}_i)$ and assume that $\bm{\Phi}$ is invertible (for example, $\bm{y}_i = \bm{x}_i$ and a strictly positive definite kernel). The role of these basis functions will be to approximate the operator $L$. We seek $\hat{u} \in V_Y$ of the form
\begin{equation}
	\hat{u}(\bm{x}) = \sum_{i=1}^n \lambda_i \phi_i(\bm{x}) = \bm{\phi}(\bm{x})^T \bm{\lambda}, \qquad \bm{\phi}(\bm{x}) := [\phi_1(\bm{x}), \dots, \phi_n(\bm{x})]^T,
\end{equation}
that interpolates $u$ on the stencil, $\hat{u}(\bm{x}_i) = u(\bm{x}_i)$ for $i = 1, \dots, n$. This leads to the form $\bm{\Phi} \bm{\lambda} = \bm{u}$ for $\bm{u} = [ u(\bm{x}_1), \dots, u(\bm{x}_n) ]^T$. We can solve for $\bm{\lambda}$ to obtain:
\begin{equation*}
	\hat{u}(\bm{x}) = \sum_{i=1}^n (\bm{\Phi}^{-1} \bm{u})_i \, \phi_i(\bm{x}) = \bm{\phi}(\bm{x})^T \bm{\Phi}^{-1} \bm{u}.
\end{equation*}
Applying $L$ (which acts on the argument $\bm{x}$ of $\bm{\phi}$) and evaluating at $\bm{z}$, the approximation of $Lu$ at $\bm{z}$ is
\begin{equation}
	Lu (\bm{z}) \approx L \hat{u} (\bm{z}) = (L \bm{\phi})(\bm{z})^T \bm{\lambda} = (L \bm{\phi})(\bm{z})^T \bm{\Phi}^{-1} \bm{u} =: \bm{w}^{L, X}(\bm{z})^T \bm{u},
\end{equation}
where $(L \bm{\phi})(\bm{z}) := [ (L\phi_1)(\bm{z}), \dots, (L\phi_n)(\bm{z}) ]^T$ and the weights $\bm{w}^{L, X}(\bm{z}) \in \mathbb{R}^n$ form a column vector.

Thus, approximating the operator reduces to a local linear solve for the weights. Transposing the definition of $\bm{w}^{L,X}(\bm{z})$ gives
\begin{equation*}
	\bm{\Phi}^T \bm{w}^{L, X}(\bm{z}) = (L \bm{\phi})(\bm{z}).
\end{equation*}
The method of undetermined coefficients on the basis $\{\phi_i\}$ gives the same weights: requiring exactness on each basis function, $\sum_{i=1}^n w_i \, \phi_j(\bm{x}_i) = (L\phi_j)(\bm{z})$ for $j = 1, \dots, n$, is exactly the system above. Finally, we let the basis of $V_Y$ be defined by radial basis functions, $\phi_i(\cdot):=\varphi(\| \cdot - \bm{y}_i \|)$, with a radial function $\varphi:[0,\infty) \rightarrow \mathbb{R}$ and centers $\{ \bm{y}_i \}_{i=1}^m \subseteq \Omega_{\bm{z}}$ introduced above. Thus we have defined the Radial Basis Function--generated Finite Difference method (RBF-FD).
\\
\noindent\textbf{Remark.} For $\bm{x} \in \Omega_{\bm{z}}$ let $w_{i,z} := \bigl( \bm{w}^{L,X}_z \bigr)_i$ be the weights computed with the fixed stencil $X$, and take $\theta_i = \delta_{\bm{x}_i}$. Then $L_w u = \sum_{i=1}^n w_i \, \theta_i(u) = L\hat{u}$ on $\Omega_{\bm{z}}$, that is, $L_w = L \circ I_X$ where $I_X u := \hat{u}$ is the interpolation operator. If $u \in V_Y$, then $I_X u = u$ and hence $L_w u = Lu$. The exactness assumption of the previous section holds with $\mathcal{V}_w = V_Y$ (working on $\Omega_{\bm{z}}$ in place of $\Omega$ and assuming $V_Y \subseteq \mathcal{N}$).

\subsection{Polynomial Augmentation}

For conditionally positive definite kernels, we augment them through the form:

\begin{align}
	&\hat{u}(\bf{x}) = \sum_{j=1}^n \lambda_i \phi_i(\bf{x}) + \sum_{j=1}^l \beta_j p_l(\bf{x}) \\	
	&\sum_{i=1}^n \lambda_i p_j(\bm{x_i}) = 0 \text{, } \forall i=1,\hdots,n
\end{align}
We can generate a Gram matrix equation as follows:
\begin{equation*}
	\begin{bmatrix}
		\bm{\hat{u}} \\ \bm{0}
	\end{bmatrix} = 
	\begin{bmatrix}
		\bm{\Phi} & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}
	\begin{bmatrix}
		\bm{\lambda} \\ \bm{\alpha}
	\end{bmatrix}
\end{equation*}
Hence, we can approximate the operator as below:
\begin{equation}\
	(L\hat{u})(\bm{z}) =
	\begin{bmatrix}
		\bm{(L\phi)(\bm{z})} \\ \bm{(Lp)(\bm{z})}
	\end{bmatrix}^T 
	\begin{bmatrix}
		\bm{\Phi} & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}^{-1}
	\begin{bmatrix}
		\bm{\hat{u}} \\ \bm{0}
	\end{bmatrix} :=
	\begin{bmatrix}
		\bm{w}_z \\ \bm{\nu}_z
	\end{bmatrix}^T 
	\begin{bmatrix}
		\bm{\hat{u}} \\ \bm{0}
	\end{bmatrix} =
	\bm{w}^T_z \bm{\hat{u}}
	\label{eq:operator_approx}
\end{equation}
Thus, we get the local augmented operator reproduction scheme:
\begin{equation}
	\begin{bmatrix}
		\bm{(L\phi)(\bm{z})} \\ \bm{(Lp)(\bm{z})}
	\end{bmatrix} =
	\begin{bmatrix}
		\bm{\Phi}^T & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix} \begin{bmatrix}
		\bm{w}_z \\ \bm{\nu}_z
	\end{bmatrix}
	\label{eq:weight_solve}
\end{equation}
The polynomial constraint setup can be viewed as a Lagrange multiplier to a minimization scheme for the weights. By \cite{Davydov2020}, given the basis for $V_Y$, $(L\bm{p})(\bm{z}) \in Range(\bm{P}^T)$ $\iff$ Eq. (\ref{eq:weight_solve}) is uniquely solvable, for any set $X$. Additionally, by \cite{Davydov2020}, Eq. (\ref{eq:operator_approx}) hold $\iff$ Eq. (\ref{eq:weight_solve}) holds. Note that depending on the the basis $V_Y$, we get the degree of $p \in \Pi_{k-1}^d$. Below are common RBFs and their respective cpd orders:
\begin{table}[ht]
    \centering
    \caption{Common radial basis functions and their minimal CPD order.}
    \label{tab:rbf-cpd-order}
    \begin{tabular}{lll}
        \hline
        \textbf{RBF} & \textbf{Radial function $\phi(r)$} & 
        \textbf{CPD order $k$} \\
        \hline
        Gaussian & $\exp(-\varepsilon^2 r^2)$ & $0$ \\[2pt]
        Inverse multiquadric & 
        $(r^2+\varepsilon^2)^{-1/2}$ & $0$ \\[2pt]
        Multiquadric & 
        $(r^2+\varepsilon^2)^{1/2}$ & $1$ \\[2pt]
        Linear & $r$ & $1$ \\[2pt]
        Cubic & $r^3$ & $2$ \\[2pt]
        Thin-plate spline & $r^2\log r$ & $2$ \\[2pt]
        Quintic & $r^5$ & $3$ \\[2pt]
        \hline
    \end{tabular}
\end{table}
This order provides us both, the degree of the polynomials $k-1$ and the number of terms:
\begin{equation*}
l=\begin{pmatrix}
	d+k-1 \\ k-1
\end{pmatrix}
\end{equation*}
As shown in \cite{Bayona2019}, a polynomial augmentation can be added with a larger degree than required for the RBF, acting as a correction term while inheriting the non-oscillatory behavior of the RBF at the boundary $\partial \Omega$.

\subsection{Consistency Error Analysis and Local System Derivation}

We begin by defining the native space of solutions for (\ref{eq:PDE}) with (\ref{eq:neumann-bc}). Suppose $\mathcal{F_K} = H^{s}(\Omega)$ with pointwise bounded evaluations. This defines an RKHS for functions $u \in H^{s}(\Omega)$, $Lu \in H^{s-k}(\Omega)$. We can also define a \textit{Kernel} $K: \Omega \times \Omega \rightarrow \mathbb{R}$ such that:
\begin{align*}
	f(\bm{z}) &= \left\langle f, K(\cdot, \bm{z}) \right\rangle_{\mathcal{F_K}}\\
	(Lf)(\bm{z}) &= \left\langle f, L_x K(\cdot, \bm{z}) \right\rangle_{\mathcal{F_K}}\\
	\bm{w}^T_z \bm{f} &= \left\langle f, \bm{w}^T_z \bm{k(\cdot)} \right\rangle_{\mathcal{F_K}}\\
	\xi_{w,L,z}(f) :&= Lf(z) - \bm{w}^T_{\bm{z}} \bm{f}
\end{align*}
According to \cite{Davydov2025}, we have the following proposition:
\begin{proposition}
(Davydov, 2025)Let D be a differential operator of order $k$, and $K$ a cpd kernel of order s on $\Omega \subseteq \mathbb{R}^d$ and $\bm{X}=\{ x_1, \dots, x_n\}$, assume that
\begin{enumerate}
\item
$\partial^{\alpha,\beta}K(\bm{z},\bm{z})$ exists for all $|\alpha|, |\beta| \le k$
\item
$L_zK(\bm{z},\bm{x}_i)$ exists for all $i=1,\dots,N$ and
\item
The equation \ref{eq:weight_solve} is consistent
\end{enumerate}
Then the kernel-based numerical differentiation formula  satisfies:
\begin{equation}
\left| \xi_{w^*,L,z}(f) \right| = \left| Lf(\bm{z}) -\sum_{i=1}^N w_{z,i}^*f(\bm{x_i}) \right| \le \mathcal{P}_{K,L,X}^2(z) ||f||_\mathcal{F_K}
\end{equation}
\end{proposition}
\noindent We note that $L$ is hence, a functional over $\mathcal{F_K}$. $L_z$ indicates applying the operator acted on the variable $\bm{z}$ in the kernel. We define $\bm{k}(\bm{z}) = [K(\bm{z},\bm{x}_1), \hdots, K(\bm{z},\bm{x}_n)]^T$. Then $\xi_{w,L,z}(f) = \left \langle f, v_{w,L,z} \right \rangle$, with $v_{w,L,z} = L_z K(\cdot, \bm{z}) - \bm{w}^T_z \bm{k}(\cdot)$. By assuming point-wise error bounds, $\| \xi_{w,L,z} \|_\mathcal{F_K^*}$,  with error estimate $\| \xi_{w,L,z} \|_\mathcal{F_K^*} = ||v_{w,K,L,X,z}||_\mathcal{F_K}$. By \cite{Davydov2025}, the functional is continuous on $\mathcal{F_K}$ if $L_xK$ exists and both, polynomial consistency and operator exactness are guaranteed. Finally, We define the power function:
\begin{equation}
\mathcal{P}_{K,L,X}^2(z) = \min \left \{||v_{w,K,L,X,z}||_\mathcal{F_K}^2: \bm{w}\in \mathbb{R}^n \text{, } \xi_{w,L,z} p = 0 \text{ for } p \in \Pi_{k-1}^d \right \}
\end{equation}
The RKHS norm can be simplified as
\begin{align*}
	||v_{w,K,L,X,z}||_\mathcal{F_K}^2 &= \left\langle v_{w,K,L,X,z}, v_{w,K,L,X,z} \right\rangle_{\mathcal{F_K}}\\
	&= \left. \xi_{w,L,y} \xi_{w,L,x} K(y,x) \right|_{y=z=x}\\
	&= \left\langle L_z K(\cdot, \bm{z}) - \bm{w}^T_{\bm{z}} \bm{k}(\cdot), L_z K(\cdot, \bm{z}) - \bm{w}^T_{\bm{z}} \bm{k}(\cdot) \right\rangle_{\mathcal{F_K}}\\
	&= \left\langle L_z K(\cdot, \bm{z}), L_z K(\cdot, \bm{z}) \right\rangle_{\mathcal{F_K}} - 2 \left\langle L_z K(\cdot, \bm{z}) , \bm{w}^T_{\bm{z}} \bm{k}(\cdot) \right\rangle_{\mathcal{F_K}} \allowbreak{} + \left\langle \bm{w}^T_{\bm{z}} \bm{k}(\cdot), \bm{w}^T_{\bm{z}} \bm{k}(\cdot) \right\rangle_{\mathcal{F_K}}\\
	&= \left. L_y L_x K(y,x) \right|_{y=z=x} - 2 \sum_{i=1}^n w_{i,\bm{z}} L_z K(\bm{z}, \bm{x}_i) + \sum_{i,j=1}^n w_{i,\bm{z}}w_{i,\bm{z}} K(\bm{x}_i, \bm{x}_j)\\
	&= 2 \left( \frac{1}{2} \left. L_y L_x K(y,x) \right|_{y=z=x} - \sum_{i=1}^n w_{i,\bm{z}} L_z K(\bm{z}, \bm{x}_i) + \frac{1}{2}\sum_{i,j=1}^n w_{i,\bm{z}}w_{i,\bm{z}} K(\bm{x}_i, \bm{x}_j) \right)
\end{align*}
Minimizing this norm for $w \in \mathbb{R}^n$ with polynomial constraints leads to Eq (\ref{eq:weight_solve}). An error estimate for functions in $\mathcal{F_K}$ of the form:
\begin{align*}
	|\xi_{w,L,z}(f)| &\le \| \xi_{w,L,z} \|_\mathcal{F_K^*} ||f||_\mathcal{F_K} \\
	\inf_{\substack{\bm{w}\in \mathbb{R}^n \\ \xi (\Pi_{k-1}^d) = 0}} |\xi_{w,L,z}(f)| &\le \inf_{\substack{\bm{w}\in \mathbb{R}^n \\ \xi (\Pi_{k-1}^d) = 0}} \| \xi_{w,L,z} \|_\mathcal{F_K^*} ||f||_\mathcal{F_K}\\	
	|\xi_{w^*,L,z}(f)| &\le \inf_{\substack{\bm{w}\in \mathbb{R}^n \\ \xi (\Pi_{k-1}^d) = 0}} ||v_{w,K,L,X,z}||_\mathcal{F_K} ||f||_\mathcal{F_K}\\
	|\xi_{w^*,L,z}(f)| &\le \mathcal{P}_{K,L,X}(\bm{z}) ||f||_\mathcal{F_K}
\end{align*}

\subsection{Growth Function}

To define the consistency order, we define the following stencil as the stencil set.
\begin{align*}
	S_{\bm{z},X} := \bigcup_{i=1}^N[\bm{z}, \bm{x}_i]
\end{align*}
Then for $U: S_{\bm{x},X} \times S_{\bm{x},X} \rightarrow \mathbb{R}$, we define a binary H\"{o}der continuity semi-norm in the form:
\begin{align*}
	|U|_{\bm{z},X,\gamma}:=\sup_{\substack{\bm{x},\bm{y}\in S_{\bm{z},X} \\ \bm{x} \neq \bm{z}, \bm{y} \neq \bm{z}}} \frac{\left| U(\bm{x},\bm{y}) - U(\bm{x},\bm{z}) - U(\bm{z},\bm{y}) + U(\bm{z},\bm{z}) \right|}{\| \bm{x} - \bm{z}\|_2^\gamma \| \bm{y} - \bm{z} \|_2^\gamma} , 0 < \gamma, \le 1
\end{align*}
Then, for $r \in \mathbb{N}$, we get
\begin{align*}
	|U|_{\bm{z},X,r+\gamma}:=\frac{1}{(\gamma+1)^2\hdots (\gamma+r)^2} \left( \sum_{|\alpha|=r} \sum_{|\beta|=r} \begin{pmatrix}	r \\ \alpha \end{pmatrix}
\begin{pmatrix}	r \\ \beta \end{pmatrix} |\partial^{\alpha,\beta}U|_{\bm{z},X,\gamma}^2	 \right)^{1/2},
\end{align*}
with the following coefficient definitions: $\begin{pmatrix}	r \\ \alpha \end{pmatrix} = \frac{r!}{\alpha!}, \alpha!=\alpha_1! \hdots \alpha_d!$.
Growth function is defined as follows:
\begin{equation}
	\rho_{k-1,D}(\bm{z},X,\mu) := \sup \left \{ Lp(\bm{z}): p\in \Pi_{k-1}^d, | p(\bm{x_j}) | \le \| \bm{x}_j - \bm{x} \|_2^\mu, j=1,\hdots,n) \right \}
\end{equation}
By \cite{Davydov2018}, the dual form of $\rho$ takes the form
\begin{equation*}
	\rho_{k-1,D}(\bm{z},Z,\| \cdot \|) := \inf{\left \{ \| \bm{w} \|: \bm{w} \in \mathbb{R}^n, \xi_{w,L,z} p = 0, p \in \Pi_{k-1}^d  \right \}}
\end{equation*}
We finally have the following by \cite{Davydov2025}:
\begin{theorem}
(Davydov, 2025) Let L be a differential operator of order $k$ and $K$ a cpd kernel of order $s$ on $\Omega$. Given $\bm{z}\in \Omega$ and $\bm{X}=\{ \bm{x}_1, \dots, \bm{x}_n \}$, with $S_{\bm{z},\bm{X}} \subset \Omega$. Assume that \ref{eq:weight_solve} is consistent and, for some $r \ge k$ and $\gamma \in (0,1]$, the kernel $K$ possesses continuous partial derivatives $\partial^{\alpha, \beta}K(\bm{x},\bm{y})$ for all $|\alpha|,|\beta| \le r$ and all $\bm{x},\bm{y} \in S_{\bm{z},\bm{X}}$, and $|K|_{\bm{z},\bm{X},r+\gamma} < \infty$. Define $q=\max\{ s, r+1 \}$. Then for $f \in \mathcal{F_K}$ the kernel-based numerical differentiation satisfies the error bound:
\begin{equation}
\left| \xi_{w^*,L,z} \right| = \left| Lf(\bm{z}) -\sum_{i=1}^N w_{z,i}^*f(\bm{x_i}) \right| \le \rho_{q, L}(\bm{z}, \bm{X}, r+\gamma) |K|_{\bm{z},\bm{X},r+\gamma} || f ||_{\mathcal{F_K}}
\end{equation}
\end{theorem}

\section{Local Least Square for RBF-FD Weights}

\subsection{Weight Constraint Setup}

Rather than assuming a form for the solution, our goal is to approximate the operator through weights dual to $u$ sampled at nodes. Suppose $m > n$ (for the case of $m < n$, refer to \cite{Slak}), thus more center nodes than stencil nodes per stencil. Let $\phi_j(\bm{z}) = K(\bm{z}, \bm{y}_j)$. The following notation for the systems of equations will be used, where $X$ is the stencil nodes and $Y$ is the center nodes:
\begin{align*}
K_{XY} &= [K(\bm{x}_i, \bm{y}_j)]_{i,j}\\
K_{YX} &= [K(\bm{x}_i, \bm{y}_j)]_{j,i}\\
K_{XX} &= [K(\bm{x}_i, \bm{x}_j)]_{i,j}\\
K_{YY} &= [K(\bm{y}_i, \bm{y}_j)]_{i,j}
\end{align*}
 Additionally, instead of assuming exactness of the form $L_zK(\bm{y}_j,\bm{z}) = \sum_{i=1}^n w_{i,\bm{z}} K(\bm{y}_j,\bm{y}_j)$ for $j = 1, \hdots, m$ as in the classical RBF-FD, we define a residual function:
\begin{align*}
	r_{j,z}(\bm{w}) :&= L_zK(\bm{y_j},\bm{z}) - \sum_{i=1}^n w_{i,\bm{z}} K(\bm{y}_j,\bm{y}_i)\\
	r_{\bm{z}} &= L_zK(Y, \bm{z}) - K_{YX} \bm{w}_{\bm{z}}, \text{ } \bm{w}: \mathbb{R}^d \rightarrow \mathbb{R}^n \text{ for } \bm{z} \in \Omega.
\end{align*}

\subsection{Consistency Error Analysis through Center-Based Kernel}

Similar to the local interpolation scheme to derive the error functional, we begin by defining the kernel used for the reproduction scheme. Rather than using the interpolation kernel $K$, we define the following kernel:
\begin{equation}
	\tilde{K}(\bm{z}_1,\bm{z}_2) := \sum_{i=1}^m K(\bm{z}_1, \bm{y}_i) K(\bm{y}_i, \bm{z}_2) = K(\bm{z}_1, Y) K(Y, \bm{z}_2)
\end{equation}
We can show this is a positive semi-definite kernel and positive definite onto the set of nodes $X$, as long as $n<=m$. (NEED TO SHOW)
\begin{align*}
	\xi_{w,L,\tilde{K},X,Y,z}(f) = \xi_{w,z}(f) :&= Lf(z) -\sum_{i=1}^n w_{i,z} f(\bm{x}_i)\\
	&= L_z \left \langle f, \, K(\cdot, Y) K(Y, \bm{z}) \right \rangle_\mathcal{F_K} - \sum_{i=1}^n w_{i,z} \left \langle f, \, K(\cdot, Y) K(Y, \bm{x_i}) \right \rangle_\mathcal{F_K}\\
	&=  \left \langle f, \, K(\cdot, Y) L_z K(Y, \bm{z}) - \sum_{i=1}^n w_{i,z} K(\cdot, Y) K(Y, \bm{x_i}) \right \rangle_\mathcal{F_K}\\
	&=  \left \langle f, \, K(\cdot, Y) L_z K(Y, \bm{z}) - K(\cdot, Y) K(Y, X) \bm{w}_{z} \right \rangle_\mathcal{F_K}\\
	&:=  \left \langle f, \, v_{w,L,\tilde{K},X,Y,z} \right \rangle_\mathcal{F_K}, \\ \text{ with } v_{w,L,\tilde{K},X,Y,z} &:= K(\cdot, Y) L_z K(Y, \bm{z}) - K(\cdot, Y) K(Y, X) \bm{w}_{z}
\end{align*}
Then, in the same way as in the interpolation case, the goal is to minimize the Riesz representer $v_{w,L,\tilde{K},X,Y,z} = v_{w,z}$. Since $\tilde{K}$ is a kernel, standard interpolation theory applies to get error estimates. However, rather than working with establish theory, we proceed by exploiting the kernel structure.
\begin{align*}
	\| v_{w,z} \|_{\mathcal{K}}^2 = \left \langle v_{w,z}, v_{w,z} \right \rangle_{\mathcal{F_K}}
\end{align*}

\subsection{Consistency Error Derivation through Standard RBF-FD}

We use the solution of the interpolation scheme to compare with the ordinary least squares, where $K_{XX} \bm{w}^*_z = L_zK(Y,\bm{z}):=\bm{b}$ is the local interpolation scheme. Thus, we can say $\bm{w}^{*T} K_{XX} \bm{w}^* = \bm{b}^T K_{XX}^{-1} \bm{b}$ since $K_{XX}$ is symmetric. Similar to the interpolation scheme, we define the error analysis and have the form:
\begin{align*}
||v_{w,K,L,X,z}||_\mathcal{F_K}^2 &= \left. L_y L_z K(y,z) \right|_{y=z=x} - 2 \sum_{i=1}^n w_{i,z} L_z K(\bm{z}, \bm{x}_i) + \sum_{i,j=1}^n w_{i,z}w_{j,z} K(\bm{x}_i, \bm{x}_j)\\
&= \left. L_y L_z K(y,z) \right. - 2 \bm{w}^T \bm{b} + \bm{w}^T K_{XX} \bm{w}\\
&= \left. L_y L_z K(y,z) \right. - 2 \bm{w}^T \bm{b} + \bm{w}^T K_{XX} \bm{w} + \bm{w}^{*T} K_{XX} \bm{w}^* - \bm{b}^T K_{XX}^{-1} \bm{b}\\
&= \left. L_y L_z K(y,z) \right. - \bm{b}^T \bm{w}^* - 2 \bm{w}^T \bm{b} + \bm{w}^T K_{XX} \bm{w} + \bm{w}^{*T} K_{XX} \bm{w}^*\\
&= \left. L_y L_z K(y,z) \right. - 2 \bm{b}^T \bm{w}^* + \bm{w}^{*T} K_{XX} \bm{w}^* - 2 \bm{w}^T \bm{b} + \bm{w}^T K_{XX} \bm{w} + \bm{w}^{*T} K_{XX} \bm{w}^*\\
&= \mathcal{P}_{K,L,X}^2(\bm{z}) - \bm{w}^T K_{XX} \bm{w}^{*} - \bm{w}^{*T} K_{XX} \bm{w} + \bm{w}^T K_{XX} \bm{w} + \bm{w}^{*T} K_{XX} \bm{w}^*\\
&= \mathcal{P}_{K,L,X}^2(\bm{z}) +  \left( \bm{w} - \bm{w}^* \right)^T K_{XX} \left( \bm{w} - \bm{w}^* \right)\\
\end{align*}
We can now bound the second term by considering $\bm{w}\rightarrow \bm{w_{LS}}$ obtained from a least squares result: 
\begin{align*}
K_{XY}K_{YX}\bm{w_{LS}}&=K_{XY}\bm{b}\\
K_{XY}K_{YX}\bm{w_{LS}}-K_{XY}K_{YX}\bm{w}^*&=K_{XY}\bm{b}-K_{XY}K_{YX}\bm{w^*}\\
K_{XY}K_{YX}(\bm{w_{LS}}-\bm{w}^*)&=K_{XY}(\bm{b}-K_{YX}\bm{w}^*)\\
(\bm{w_{LS}}-\bm{w}^*)&= K_{YX}^{\dagger}\bm{r}^*\\
|| \bm{w_{LS}}-\bm{w}^* ||_2 &\le ||K_{YX}^{\dagger} ||_2 || \bm{r}^* ||_2\\
|| \bm{w_{LS}}-\bm{w}^* ||_2 &\le \frac{|| \bm{r}^* ||_2}{\sigma_{min}(K_{XY})}
\end{align*}
Finally, we say that $r^*_j=v_{w^*,K,L,X,z}(\bm{y}_j)$. Then $|| \bm{r}^* ||^2_2 = || v_{w^*,K,L,X,z}(Y) ||_2^2$. Applying kernel evaluation on $v_{w^*,K,L,X,z}$ and the power function for interpolations onto $v$ we get:
\begin{align*}
|| \bm{r}^* ||_2^2 &= \sum_{i=1}^m v_{w^*,K,L,X,z}^2(y_i)\\
&\le \sum_{i=1}^m || v_{w^*,K,L,X,z} ||^2_\mathcal{F_K} \mathcal{P}_{K,I_X,X}^2(\bm{y_i})\\
&\le \mathcal{P}_{K,L,X}^2(\bm{z}) \sum_{i=1}^m \mathcal{P}_{K,I_X,X}^2(\bm{y_i})
\end{align*}
Therefore, the error estimate (\ref{eq:LeastSquaresEstimate}) for the local Ordinary Least Squares behaves as follows:
\begin{align}
||v_{w_{LS},K,L,X,z}||_\mathcal{F_K}^2 &\le \mathcal{P}_{K,L,X}^2(\bm{z}) +  \lambda_{max}(K_{XX}) || \bm{w} - \bm{w}^* ||_2^2 \nonumber \\
&\le \mathcal{P}_{K,L,X}^2(\bm{z}) + \lambda_{max}(K_{XX}) \frac{\mathcal{P}_{K,L,X}^2(\bm{z}) \sum_{i=1}^m \mathcal{P}_{K,I_X,X}^2(\bm{y_i})}{\sigma_{min}^2(K_{XY})} \nonumber \\
&\le \mathcal{P}_{K,L,X}^2(\bm{z}) \left(1 + \frac{\lambda_{max}(K_{XX})\sum_{i=1}^m \mathcal{P}_{K,I_X,X}^2(\bm{y_i})}{\sigma_{min}^2(K_{XY})} \right)
\label{eq:LeastSquaresEstimate}
\end{align}

\subsection{Least Squares Finite Spaces and Interpolation Results}

Suppose $\left\{p_i\right\}_{i=1}^l$ is a set of polynomials augmenting the radial basis functions. Let $\Pi_k=span\left\{p_i\right\}_{i=1}^l$ with polynomial degree up to $k$. We want to reproduce the operator exactly on the polynomial basis: 
\begin{align*}
	Lp_j(\bm{z}) :&= \sum_{i=1}^n w_{i,z} p_j(\bm{x_i}) \text{ for } j = 1, \hdots, l\\
	L\bm{p}(\bm{z}) &= \bm{P^T} \bm{w}_z
\end{align*}
where $\bm{P}_{ij} := p_j(\bm{x}_i) \in \mathbb{R}^{n \times l}$. Thus, we have a constrained least squares problem in the form:
\begin{equation}
	\left\{
	\begin{aligned}
		&\min J(\bm{w}_z) := \min \frac{1}{2}\|\bm{r}(\bm{w}_z)\|_2^2 \quad \text{for } \bm{x} \in \Omega \\
		&\text{subject to } K_{YX} \bm{w}_z = L\bm{p}(\bm{z})
	\end{aligned}
	\right.
\end{equation}
We can solve this setup by defining the Lagrangian $\mathcal{L}$ for the problem as follows:
\begin{align*}
\mathcal{L}(\bm{w},\bm{\nu})
&:= J(\bm{w})
+ \left\langle \bm{\nu},\,\bm{P}^{T}\bm{w} - L\bm{p}(\bm{z}) \right\rangle \\
&= \frac{1}{2}
\left\langle K_{YX}\bm{w},\,K_{YX}\bm{w} \right\rangle
- \left\langle K_{YX}\bm{w},\,L_z K(Y, \bm{z}) \right\rangle
+ \frac{1}{2}
\left\langle L_z K(Y,\bm{z}),\,L_z K(Y,\bm{z}) \right\rangle \\
&\qquad
+ \left\langle \bm{\nu},\,\bm{P}^{T}\bm{w} - L\bm{p}(\bm{z}) \right\rangle.
\end{align*}
The resulting KKT equations for this system are then:
\begin{align*}
	\nabla_w \mathcal{L} &= 0: K_{XY} L_z K(Y,\bm{z}) = K_{XY} K_{YX} \bm{w} + \bm{P} \bm{\nu} \label{eq:normal_eq}\\
	\nabla_\nu \mathcal{L} &= 0: \bm{P}^T \bm{w} = L\bm{p}
	%\label{eq:constraint}
\end{align*}
The augmented matrix then takes the form:
\begin{equation}
	\begin{bmatrix}
		K_{XY} K_{YX} & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}
	\begin{bmatrix}
		\bm{w} \\ \bm{\nu}
	\end{bmatrix} =
	\begin{bmatrix}
		X_{XY} L_z K(Y,\bm{z}) \\ L\bm{p}(\bm{z})
	\end{bmatrix}
	\label{eq:augmented}
\end{equation}
From \cite{Davydov2020}, Equation (\ref{eq:augmented}) has a unique solution given $L\bm{p} \in Range(\bf{P}^T)$. The primary goal is to approximate $Lu(\bm{z}) \approx \bm{w}^T_z \bm{u}$, with $\bm{u}=\left[ u(\bm{x}_1), \hdots, u(\bm{x}_n) \right]^T$. Define $L_w f(\bm{z}) := \bm{w}^T_z \bm{f}$ for some function $f$. Thus, we have the following:
\begin{align*}
	L_w f(\bm{z}) &= \bm{w}^T_z \bm{f}\\
			&= \begin{bmatrix}
					\bm{w}^T_z & \bm{\nu}^T
				\end{bmatrix}
				\begin{bmatrix}
					\bm{f} \\ \bm{0}
				\end{bmatrix}\\
			&= \left(
	\begin{bmatrix}
		K_{XY} K_{YX} & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}^{-1}
	\begin{bmatrix}
		K_{XY} L_z K(Y,\bm{z}) \\ L\bm{p}(\bm{z})
	\end{bmatrix}
				\right)^T 
				\begin{bmatrix}
					\bm{f} \\ \bm{0}
				\end{bmatrix}\\
			&= \begin{bmatrix}
					L_z K(\bm{z},Y) K_{XY} & L\bm{p}^T(\bm{z})
				\end{bmatrix}
	\begin{bmatrix}
		K_{XY} K_{YX} & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}^{-T}
				\begin{bmatrix}
					\bm{f} \\ \bm{0}
				\end{bmatrix}\\
			&= \begin{bmatrix}
					L_z K(\bm{z},Y) K_{XY} & L\bm{p}^T(\bm{z})
				\end{bmatrix}
				\begin{bmatrix}
					\bm{q} \\ \bm{\alpha}
				\end{bmatrix}\\
			&= L_z K(\bm{z},Y) \left(K_{YX} \bm{q}\right) + L\bm{p}^T (\bm{z}) \bm{\alpha}
\end{align*}
Next, we can show that the coefficients $\bm{q}, \bm{\alpha}$ relate to the coefficients of $u$ as follows:
\begin{equation*}
	\begin{bmatrix}
		\bm{f} \\ \bm{0}
	\end{bmatrix} = 
	\begin{bmatrix}
		K_{XY} K_{YX} & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}
	\begin{bmatrix}
		\bm{q} \\ \bm{\alpha}
	\end{bmatrix}
\end{equation*}
Define $V_m = \left\{ \rho(z) : \rho(\bm{z}) = \sum_{i=1}^m \lambda_i K(\bm{z},\bm{y}_i) \text{ and } \lambda \in Range\left(K_{YX} \right), \bm{y}_i \in Y \right\} \subseteq V_Y$. Let $s \in V_m + \Pi_k$. Thus, $s(\bm{z}) = K(\bm{z},Y)K_{XY} q + \bm{p}^T(\bm{z}) \bm{\alpha}$ where $\bm{q} \in \mathbb{R}^{n}$ and $\bm{\alpha} \in \mathbb{R}^{l}$. Then:
\begin{align*}
	\bm{s} = s(X) &= K_{XY} K_{YX} \bm{q} + \bm{P} \bm{\alpha} = f(X) = \bm{f}\\
	\bm{P}^T \bm{q} &= \bm{0}
\end{align*}
Hence, we get interpolation of $f$ at the node set $X$. Furthermore, from the KKT matrix we can show that $K_{XY} \bm{r} = \bm{P} \bm{\nu}$ which implies that $K_{XY} \bm{r} \in range(\bm{P}) = \left( ker \left(\bm{P}^T \right) \right)^\perp$. Thus, we get the relationship:
\begin{equation}
\bm{r}^T \left( K_{YX} \bm{q} \right)=\bm{0} \text{ for } \bm{P}^T \bm{q} = \bm{0} \label{eq:res_orth}
\end{equation}
Finally, we check the consistency error for $s \in V_h+\Pi_k$ with $\bm{P}^T \bm{q}=\bm{0}$.
\begin{align*}
	Ls(\bm{z})-L_hs(\bm{z}) &= L\left[K(\bm{z},Y)K_YX \bm{q} + \bm{p}^T(\bm{z}) \bm{\alpha}\right] - \bm{w}^T_z\bm{s}\\
	&= L_z K(\bm{z},Y)K_{YX} \bm{q} + L\bm{p}^T(\bm{z}) \bm{\alpha} - \bm{w}^T_z(K_{XY} K_{YX} \bm{q} + \bm{P} \bm{\alpha})\\
	&= L_z K(\bm{z},Y)K_{YX} \bm{q} + L\bm{p}^T(\bm{z}) \bm{\alpha} - \bm{w}^T_z K_{XY} K_{YX} \bm{q} - \bm{w}^T_z \bm{P} \bm{\alpha}\\
	&= \left[ L_z K(\bm{z},Y) - K_{YX} \bm{w}_z \right]^T K_{YX} \bm{q} + \left[ L\bm{p}(\bm{z}) - \bm{P}^T \bm{w}_z \right]^T \bm{\alpha}\\
	&= \bm{r}^T K_{YX} \bm{q} + \left[ L\bm{p}(\bm{z}) - \bm{P}^T \bm{w}_z \right]^T \bm{\alpha}\\
\end{align*}
Hence, we conclude $Ls(\bm{z})-L_hs(\bm{z}) = 0 \text{ due to (\ref{eq:augmented}) and (\ref{eq:res_orth}}) \label{eq:repro}$.
