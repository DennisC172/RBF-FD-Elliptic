\section{Problem Setting}

Following the standard RBF-FD methods, let $u \in \mathcal{F}$ be a solution for a PDE in form of:
\begin{equation}
	- Lu = -\diver\!\big(A \grad u\big) = f \quad \text{in } \Omega,
	\label{eq:PDE}
\end{equation}

Let $1<p<\infty$, $A \in W^{1,\infty} (\bar{\Omega}; \mathbb{R}^{d \times d}):\bar{\Omega} \rightarrow \mathbb{R}^{d \times d}$ is an symmetric positive definite anisotropic tensor, where
\begin{equation}
	\zeta |\xi|^2 \le \bf{A}(\bf{x}) \xi \cdot \xi \le Z |\xi|^2 \text{, } 0 < \zeta \le Z
\end{equation}

Assume $\Omega \subset \mathbb{R}^d$ and $f \in W^{k,p}(\Omega)$ is the forcing term acting in the domain $\Omega$. The problem is subject to Neumann and Dirichlet Boundary Conditions partitioned within $\partial \Omega$ of the form:
\begin{align}
  u &= g_D \quad \text{on } \Gamma_D \subset \partial\Omega, \\
  n \cdot A \grad u &= g_N \quad
  \text{on } \Gamma_N \subset \partial\Omega,
  \label{eq:neumann-bc}
\end{align}

For simplicity, we assume $\Omega$ be a bounded $C^{1,1}(\partial \Omega)$ domain, with $\partial \Omega = \Gamma_D \cup \Gamma_N$ and $\Gamma_D \cap \Gamma_N = \emptyset$. Furthermore, let $g_D \in W^{k-1/p,p}(\partial \Omega)$, $g_N \in W^{k-1-1/p,p}(\partial \Omega)$. By \textit{Agmon-Douglis-Nirenberg} Theorem \cite{Agmon}, we have the following estimate for $u$:
\begin{equation}
\|u\|_{W^{k,p}(\Omega)} \leq C\left(\|Lu\|_{W^{k-2,p}(\Omega)} + \|g_D\|_{W^{k-1/p,p}(\partial\Omega)} + \|g_N\|_{W^{k-1-1/p,p}(\partial\Omega)} + \|u\|_{L^p(\Omega)} \right).
\end{equation}

In the case that we work on a Hilbert space, we then have for $p=2$ ($W^{k,2} := H^k$) the following estimate for $u$:
\begin{equation}
\|u\|_{H^k(\Omega)} \leq C\left(\|Lu\|_{H^{k-2}(\Omega)} + \|g_D\|_{H^{k-\frac{1}{2}}(\partial\Omega)} + \|g_N\|_{H^{k-\frac{3}{2}}(\partial\Omega)} + \|u\|_{L^2(\Omega)} \right).
\end{equation}

If we want to study cases where $(Lu)(\bm{x}) \in L^\infty(\Omega)$. we can formulate the setup by having $f \in W^{s,p}(\Omega)$ and, by Morrey Embedding, 
\begin{equation}
	W^{s,p} \hookrightarrow C^{0,\alpha} \text{, with } \alpha=s-\frac{d}{p}>0
\end{equation}
In other words, $s > \frac{d}{p}$. Suppose $p=2$, then $s > \frac{d}{2}$. Given an operator $L: H^{s+2}(\Omega) \rightarrow H^{s}(\Omega)$, we define an error functional of the form:
\begin{equation}
	\xi_{L,x}: H^{s+2}(\Omega) \rightarrow \mathbb{R} \text{, } \xi_{L,x} \in \left(H^{s+2}\right)^*(\Omega)
\end{equation}

Then by Riesz Representation Theorem, there exists a $v_{L,x} \in H^{s+2}(\Omega)$ such that $\xi_{L,x} (u) = \left\langle u , v_{L,x} \right\rangle_{H^{s+2}(\Omega)}$, and $\| \xi_{L,x} |\|_{\left(H^{s+2}(\Omega)\right)^*} = \| v_{L,x} \|_{H^{s+2}(\Omega)}$

\section{RBF-FD Method}

\subsection{Local Approximation setup}
To obtain an approximation of the operator $L$ at point $\bf{z}$, we can approximate $\hat{u} \approx u$ on a neighborhood $\Omega \subseteq \mathbb{R}^d$ of $\bf{z}$ such that we approximate $Lu \approx L\hat{u}$. We define the basis functions in the form $\phi_i(\bm{x}): \Omega \rightarrow \mathbb{R}$ for $i = 1, \hdots , m$. Define the vector space $V_Y = span\left\{ \phi_1, \dots, \phi_m \right\}$. Let $X = \left\{ \bm{x}_1, \hdots, \bm{x}_n \right\} \subseteq \Omega$ be a set of stencil nodes for a point $\bf{z}$ in $\Omega$. For the classical approximation scheme, let $m=n$. We can now define $\bm{\Phi} \in \mathbb{R}^{n \times m}$, where $\bm{\Phi}_{ij} = \phi_j(\bm{x}_i)$. The role of these basis functions will be to approximate the operator $L$. The approximation of $u$ takes the form of:
\begin{equation}
	\hat{u}(\bf{x}) =  \sum_{i=1}^n \lambda_i \phi_i(\bf{x})
\end{equation}

This will lead to the form $\bf{\Phi} \bf{\lambda} = \hat{\bf{u}}$ for $\hat{\bf{u}} = [ u(\bf{x_1}), \hdots, u(\bf{x_n}) ]^T.$ We can simplify for $\lambda$ to obtain:
\begin{equation}
\hat{u}(\bf{x}) = \sum_{i=1}^n (\bf{\Phi}^{-1} \bf{u})_i \phi_i(\bf{x}) = \bf{\phi}(\bf{x})^T \bf{\Phi}^{-1} \bf{u}
\end{equation}

The approximation of $Lu$ at $\bm{z}$ is in the form of:
\begin{equation}
	Lu (\bf{z}) \approx L \hat{u} (\bf{z}) = L \bf{\phi}^T(\bf{z}) \bf{\lambda} = L \bf{\phi}^T(\bf{z}) \bf{\Phi}^{-1} \bf{u} := w^{L, X}(\bf{z}) \bf{u}
\end{equation}

Thus, approximation the operator turns into local solver for the weights $w_i$ from the form:
\begin{equation}
	\bf{\Phi}^T \bf{w^{L, X}(\bf{z})} = L \bf{ \phi} (\bf{z})
\end{equation}

Through the method of undefined coefficients on the basis $\{\phi_i\}$ we get the same resulting set of weights. Finally, by defining the basis for $V_Y$ as radial basis functions, with $\phi_i(\bm{x}) := \phi(||\bm{x}-\bm{y}_i||)$ and centers $\left\{ \bm{y}_i \right\}_{i=1}^m \subseteq \Omega$. Thus we defined the Radial Basis Function generated Finite Difference method (RBF-FD). 

\subsection{Polynomial Augmentation}

For conditionally positive definite kernels, we augment them through the form:

\begin{align}
	&\hat{u}(\bf{x}) = \sum_{j=1}^n \lambda_i \phi_i(\bf{x}) + \sum_{j=1}^l \beta_j p_l(\bf{x}) \\	
	&\sum_{i=1}^n \lambda_i p_j(\bm{x_i}) = 0 \text{, } \forall i=1,\hdots,n
\end{align}

We can generate a Gram matrix equation of the form:

\begin{equation}
	\begin{bmatrix}
		\bm{\hat{u}} \\ \bm{0}
	\end{bmatrix} = 
	\begin{bmatrix}
		\bm{\Phi} & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}
	\begin{bmatrix}
		\bm{\lambda} \\ \bm{\alpha}
	\end{bmatrix}
\end{equation}

Hence, we can approximate the operator to take the form:

\begin{equation}\
	(L\hat{u})(\bm{x}) =
	\begin{bmatrix}
		\bm{(L\phi)(\bm{x})} \\ \bm{(Lp)(\bm{x})}
	\end{bmatrix}^T 
	\begin{bmatrix}
		\bm{\Phi} & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}^{-1}
	\begin{bmatrix}
		\bm{\hat{u}} \\ \bm{0}
	\end{bmatrix} :=
	\begin{bmatrix}
		\bm{w}(\bm{x}) \\ \bm{\nu}(\bm{x})
	\end{bmatrix}^T 
	\begin{bmatrix}
		\bm{\hat{u}} \\ \bm{0}
	\end{bmatrix} =
	\bm{w}^T(\bm{x}) \bm{\hat{u}}
\end{equation}

Thus, we get the local augmented operator reproduction scheme:

\begin{equation}
	\begin{bmatrix}
		\bm{(L\phi)(\bm{x})} \\ \bm{(Lp)(\bm{x})}
	\end{bmatrix} =
	\begin{bmatrix}
		\bm{\Phi}^T & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix} \begin{bmatrix}
		\bm{w}(\bm{x}) \\ \bm{\nu}(\bm{x})
	\end{bmatrix}
	\label{eq: weight_solve}
\end{equation}

The polynomial constraint setup can be viewed as a Lagrange multiplier to a minimization scheme for the weights. By \cite{Davydov2020}, given the basis for $V_Y$ and $(L\bm{p})(\bm{x}) \in Range(\bm{P}^T)$, then Eq. (\ref{eq: weight_solve}) is uniquely solvable. Note that depending on the the basis $V_Y$, we get the degree of $p \in \Pi_{k-1}^d$. Below are common RBFs and their respective cpd orders:

\begin{table}[ht]
    \centering
    \caption{Common radial basis functions and their minimal CPD order.}
    \label{tab:rbf-cpd-order}
    \begin{tabular}{lll}
        \hline
        \textbf{RBF} & \textbf{Radial function $\phi(r)$} & 
        \textbf{CPD order $k$} \\
        \hline
        Gaussian & $\exp(-\varepsilon^2 r^2)$ & $0$ \\[2pt]
        Inverse multiquadric & 
        $(r^2+\varepsilon^2)^{-1/2}$ & $0$ \\[2pt]
        Multiquadric & 
        $(r^2+\varepsilon^2)^{1/2}$ & $1$ \\[2pt]
        Linear & $r$ & $1$ \\[2pt]
        Cubic & $r^3$ & $2$ \\[2pt]
        Thin-plate spline & $r^2\log r$ & $2$ \\[2pt]
        Quintic & $r^5$ & $3$ \\[2pt]
        \hline
    \end{tabular}
\end{table}

This order provides us both, the degree of the polynomials $k-1$ and the number of terms:
\begin{equation}
l=\begin{pmatrix}
	d+k-1 \\ k-1
\end{pmatrix}
\end{equation}

As shown in \cite{Bayona2019}, a polynomial augmentation can be added with a larger degree than required for the RBF, acting as a correction term while inheriting the non-oscillatory behavior of the RBF at the boundary $\partial \Omega$.

\subsection{Consistency Error Analysis}

We begin by defining the native space of solutions for (\ref{eq:PDE}) with (\ref{eq:neumann-bc}). Suppose $\mathcal{F_K} = H^{s+2}(\Omega)$. This is defines an RKHS for functions $u \in H^{s+2}(\Omega)$, $Lu \in H^{s}(\Omega)$ and by Morrey Embedding, $L(\cdot)(\bm{x})$ is bounded, lienar functional on $H^{s+2}(\Omega)$. Thus, by Riesz, $v_{Lu,x} \in H^{s+2}(\Omega)$. We can also define a \textit{Kernel} $K: \Omega \times \Omega \rightarrow \mathbb{R}$ such that
\begin{align}
	f(x) &= \left\langle f, K(\cdot, x) \right\rangle_{\mathcal{F_K}}\\
	(Lf)(x) &= \left\langle f, L_x K(\cdot, \bm{x}) \right\rangle_{\mathcal{F_K}}\\
	\bm{w}^T(\bm{x}) \bm{f} &= \left\langle f, \bm{w}^T(\bm{x}) \bm{k(\cdot)} \right\rangle_{\mathcal{F_K}}\\
	\xi_{w,L,x}(f) :&= Lf(x) - \bm{w}^T(\bm{x}) \bm{f}\\
	\| \xi_{w,L,x} \|_\mathcal{F_K^*} :&= \sup_{\substack{f \in \mathcal{F_K} \\ ||f||_\mathcal{F_K} \le 1}} |\xi_{w,L,x}(f)|
\end{align}

$L_x$ indicates applying the operator acted on the variable $\bm{x}$ in the kernel. We define $\bm{k}(\bm{z}) = [k(\bm{z},\bm{x}_1), \hdots, k(\bm{z},\bm{x}_n)]^T$. Since $\| \xi_{w,L,x} \|_\mathcal{F_K^*} <\infty$, by Riesz Representation Theorem, $v_{w,L,x} = L_x K(\cdot, \bm{x}) - \bm{w}^T(\bm{x}) \bm{k}(\cdot)$ with error estimate $\| \xi_{w,L,x} \|_\mathcal{F_K^*} = ||v_{w,L,x}||_\mathcal{F_K}$ By \cite{Davydov2025}, the functional is continuous on $\mathcal{F_K}$ if $L_xK$ exists and both, polynomial consistency and operator exactness are guaranteed. Finally, We define the power function $\mathcal{P}^2(x) = ||v_{w,L,x}||_\mathcal{F_K}^2$. Note that:

\begin{align*}
	\mathcal{P}^2(\bm{x}) &= \left\langle v_{w,L,x}, v_{w,L,x} \right\rangle_{\mathcal{F_K}}\\
	&= \left\langle L_x K(\cdot, \bm{x}) - \bm{w}^T(\bm{x}) \bm{k}(\cdot), L_x K(\cdot, \bm{x}) - \bm{w}^T(\bm{x}) \bm{k}(\cdot) \right\rangle_{\mathcal{F_K}}\\
	&= \left\langle L_x K(\cdot, \bm{x}), L_x K(\cdot, \bm{x}) \right\rangle_{\mathcal{F_K}} - 2 \left\langle L_x K(\cdot, \bm{x}) , \bm{w}^T(\bm{x}) \bm{k}(\cdot) \right\rangle_{\mathcal{F_K}} + \left\langle \bm{w}^T(\bm{x}) \bm{k}(\cdot), \bm{w}^T(\bm{x}) \bm{k}(\cdot) \right\rangle_{\mathcal{F_K}}\\
	&= \left. L_y L_z K(y,z) \right|_{y=x=z} - 2 \sum_{i=1}^n w_i(\bm{x}) L_x K(\bm{x}, \bm{x}_i) + \sum_{i,j=1}^n w_i(\bm{x})w_j(\bm{x}) K(\bm{x}_i, \bm{x}_j)\\
	&= \left. L_y L_z K(y,z) \right|_{y=x=z} - \sum_{i=1}^n w_i(\bm{x}) L_x K(\bm{x}, \bm{x}_i)
\end{align*}

The last line is carried out by Eq (\ref{eq: weight_solve}).

\section{Weight Constraint RBF-FD Setup}

Rather than assuming a form for the solution, our goal is to approximate the operator through weights dual to $u$ sampled at nodes. Suppose $m > n$ (for the case of $m < n$, refer to \cite{Slak}), thus more center nodes than stencil nodes per stencil. Additionally, instead of assuming exactness of the form $L\phi_j(\bm{x}) = \sum_i^n w_i(\bm{x}) \phi_j(\bm{x}_i)$ for $j = 1, \hdots, m$ as in the classical RBF-FD, we define a residual function:
\begin{align}
	r_j(\bm{x}) :&= L\phi_j(\bm{x}) - \sum_{i=1}^n w_i(\bm{x}) \phi_j(\bm{x_i})\\
	r(\bm{x}) &= L\bm{\phi}(\bm{x}) - \bm{\Phi^T} \bm{w}(\bm{x}), \text{ } \bm{w}: \mathbb{R}^n \rightarrow \mathbb{R}^n \text{ for } \bm{x} \in \Omega.
\end{align}

Suppose $\left\{p_i\right\}_{i=1}^l$ is a set of polynomials augmenting the radial basis functions. Let $\Pi_k=span\left\{p_i\right\}_{i=1}^l$ with polynomial degree up to $k$. We want to reproduce the operator exactly on the polynomial basis: 
\begin{align}
	Lp_j(\bm{x}) :&= \sum_{i=1}^n w_i(\bm{x}) p_j(\bm{x_i}) \text{ for } j = 1, \hdots, l\\
	L\bm{p}(\bm{x}) &= \bm{P^T} \bm{w}
\end{align}

where $\bm{P}_{ij} := p_j(\bm{x}_i) \in \mathbb{R}^{n \times l}$. Thus, we have a constrained least squares problem in the form:
\begin{equation}
	\left\{
	\begin{aligned}
		\min J(\bm{w}(\bm{x})) &:= \min \frac{1}{2}\|\bm{r}(\bm{x})\|_2^2 \quad \text{for } \bm{x} \in \Omega \\
		\text{subject to } \bm{\Phi}^T \bm{w} &= L\bm{p}
	\end{aligned}
	\right.
\end{equation}

We can solve this setup by defining the Lagrangian $\mathcal{L}$ for the problem as follows:
\begin{align*}
\mathcal{L}(\bm{w},\bm{\nu})
&:= J(\bm{w})
+ \left\langle \bm{\nu},\,\bm{P}^{T}\bm{w} - L\bm{p}(\bm{x}) \right\rangle \\
&= \frac{1}{2}
\left\langle \bm{\Phi}^{T}\bm{w},\,\bm{\Phi}^{T}\bm{w} \right\rangle
- \left\langle \bm{\Phi}^{T}\bm{w},\,L\bm{\phi} \right\rangle
+ \frac{1}{2}
\left\langle L\bm{\phi},\,L\bm{\phi} \right\rangle \\
&\qquad
+ \left\langle \bm{\nu},\,\bm{P}^{T}\bm{w} - L\bm{p}(\bm{x}) \right\rangle.
\end{align*}

\section{Consistency Conditions for Finite Spaces and Interpolation Results}

The resulting KKT equations for this system are then:
\begin{align}
	\nabla_w \mathcal{L} &= 0: \bm{\Phi} L\bm{\phi} = \bm{\Phi} \bm{\Phi}^T \bm{w} + \bm{P} \bm{\nu} \label{eq:normal_eq}\\
	\nabla_\nu \mathcal{L} &= 0: \bm{P}^T \bm{w} = L\bm{p}
	\label{eq:constraint}
\end{align}

The augmented matrix then takes the form:
\begin{equation}
	\begin{bmatrix}
		\bm{\Phi} \bm{\Phi}^T & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}
	\begin{bmatrix}
		\bm{w} \\ \bm{\nu}
	\end{bmatrix} =
	\begin{bmatrix}
		\bm{\Phi} L\bm{\phi} \\ L\bm{p}
	\end{bmatrix}
	\label{eq:augmented}
\end{equation}

From \cite{Davydov2020}, Equation (\ref{eq:augmented}) has a unique solution given $L\bm{p} \in Range(\bf{P}^T)$. The primary goal is to approximate $Lu(\bm{x}) \approx \bm{w}^T(\bm{x}) \bm{u}$, with $\bm{u}=\left[ u(\bm{x}_1), \hdots, u(\bm{x}_n) \right]^T$. Define $L_h f(\bm{x}) := \bm{w}^T (\bm{x}) \bm{f}$ for some function $f$. Thus, we have the following:
\begin{align*}
	L_h f(\bm{x}) &= \bm{w}^T(\bm{x}) \bm{f}\\
			&= \begin{bmatrix}
					\bm{w}^T & \bm{\nu}^T
				\end{bmatrix}
				\begin{bmatrix}
					\bm{f} \\ \bm{0}
				\end{bmatrix}\\
			&= \left(
	\begin{bmatrix}
		\bm{\Phi} \bm{\Phi}^T & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}^{-1}
	\begin{bmatrix}
		\bm{\Phi} L\bm{\phi} \\ L\bm{p}
	\end{bmatrix}
				\right)^T 
				\begin{bmatrix}
					\bm{f} \\ \bm{0}
				\end{bmatrix}\\
			&= \begin{bmatrix}
					L\bm{\phi}^T \bm{\Phi}^T & L\bm{p}^T
				\end{bmatrix}
	\begin{bmatrix}
		\bm{\Phi} \bm{\Phi}^T & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}^{-T}
				\begin{bmatrix}
					\bm{f} \\ \bm{0}
				\end{bmatrix}\\
			&= \begin{bmatrix}
					L\bm{\phi}^T \bm{\Phi}^T & L\bm{p}^T
				\end{bmatrix}
				\begin{bmatrix}
					\bm{q} \\ \bm{\alpha}
				\end{bmatrix}\\
			&= L\bm{\phi}^T (\bm{x}) \left(\bm{\Phi}^T \bm{q}\right) + L\bm{p}^T (\bm{x}) \bm{\alpha}
\end{align*}

Next, we can show that the coefficients $\bm{q}, \bm{\alpha}$ relate to the coefficients of $u$ as follows:

\begin{equation}
	\begin{bmatrix}
		\bm{f} \\ \bm{0}
	\end{bmatrix} = 
	\begin{bmatrix}
		\bm{\Phi} \bm{\Phi}^T & \bm{P} \\
		\bm{P}^T & \bm{0}
	\end{bmatrix}
	\begin{bmatrix}
		\bm{q} \\ \bm{\alpha}
	\end{bmatrix}
\end{equation}

Define $V_h = \left\{ \rho(x) : \rho(x) = \sum_{i=1}^m \lambda_i \phi_i(x) \text{ such that } \lambda \in Range\left(\bm{\Phi}^T \right) \right\} \subseteq V_Y$. Let $s \in V_h + \Pi_k$. Thus, $s(x) = \bm{\phi}^T(\bm{x})\bm{\Phi}^T q + \bm{p}^T(\bm{x}) \bm{\alpha}$ where $\bm{q} \in \mathbb{R}^{n}$ and $\bm{\alpha} \in \mathbb{R}^{l}$. Then:
\begin{align}
	\bm{s} = s(X) &= \bm{\Phi} \bm{\Phi}^T \bm{q} + \bm{P} \bm{\alpha} = f(X) = \bm{f}\\
	\bm{P}^T \bm{q} &= \bm{0}
\end{align}

Hence, we get interpolation of $f$ at the node set $X$. Furthermore, from the KKT matrix we can show that $\bm{\Phi} \bm{r} = \bm{P} \bm{\nu}$ which implies that $\bm{\Phi} \bm{r} \in range(\bm{P}) = \left( ker \left(\bm{P}^T \right) \right)^\perp$. Thus, we get the relationship:
\begin{equation}
\bm{r}^T \left( \bm{\Phi}^T \bm{z} \right)=\bm{0} \text{ for } \bm{P}^T \bm{z} = \bm{0} \label{eq:res_orth}
\end{equation}

Finally, we check the consistency error for $s \in V_h+\Pi_k$ with $\bm{P}^T \bm{q}=\bm{0}$.
\begin{align*}
	Ls(x)-L_hs(x) &= L\left(\bm{\phi}^T(\bm{x})\bm{\Phi}^T \bm{q} + \bm{p}^T(\bm{x}\right) \bm{\alpha}) - \bm{w}^T(\bm{x})\bm{s}\\
	&= L\bm{\phi}^T(\bm{x})\bm{\Phi}^T \bm{q} + L\bm{p}^T(\bm{x}) \bm{\alpha} - \bm{w}^T(\bm{x})(\bm{\Phi} \bm{\Phi}^T \bm{q} + \bm{P} \bm{\alpha})\\
	&= L\bm{\phi}^T(\bm{x})\bm{\Phi}^T \bm{q} + L\bm{p}^T(\bm{x}) \bm{\alpha} - \bm{w}^T(\bm{x})\bm{\Phi} \bm{\Phi}^T \bm{q} - \bm{w}^T(\bm{x}) \bm{P} \bm{\alpha}\\
	&= \left( L\bm{\phi}(\bm{x}) - \bm{\Phi}^T \bm{w}(\bm{x}) \right)^T \bm{\Phi}^T \bm{q} + \left( L\bm{p}^T(\bm{x}) - \bm{P}^T \bm{w}(\bm{x}) \right)^T \bm{\alpha}\\
	&= \bm{r}^T \bm{\Phi}^T \bm{q} + \left( L\bm{p}(\bm{x}) - \bm{P}^T \bm{w}(\bm{x}) \right)^T \bm{\alpha}\\
\end{align*}
\begin{equation}
	Ls(x)-L_hs(x) = 0 \text{ due to (\ref{eq:constraint}) and (\ref{eq:res_orth}}) \label{eq:repro}
\end{equation}

\section{Consistency Results for Function Approximations}.

Let $u$ be a solution to (\ref{eq:PDE}) and $s \in V_h+\Pi_k$. We can say $u = s + (u - s)$ to get the following:
\begin{align*}
\mathcal{E}[u](\bm{x}) :&= Lu(\bm{x}) - L_hu(\bm{x})\\
	&= Lu(\bm{x}) - \bm{w}^T(\bm{x})\bm{u}\\
	&= L((u-s) + s)(\bm{x}) - \bm{w}^T(\bm{x})\bm{u} - L_h s(\bm{x}) + L_h s(\bm{x})\\
	&= L(u-s)(\bm{x}) + (L s(\bm{x}) - L_h s(\bm{x})) + L_h(s-u)(\bm{x})\\
	&= L(u-s)(\bm{x}) - \bm{w}^T(\bm{x})(\bm{u}-\bm{s}) \text{, by (\ref{eq:repro})}\\
	&= Le(\bm{x}) - \bm{w}^T(\bm{x})\bm{e} \text{, } e(\bm{x})=u(\bm{x})-s(\bm{x})\\
	&= \mathcal{E}[e](\bm{x})
\end{align*}
